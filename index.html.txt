        async function handleChatSubmit(event) {
            event.preventDefault();
            const inputEl = document.getElementById('chat-input');
            const message = inputEl.value.trim();
            
            if (!message) return;
            
            // Добавляем сообщение пользователя в UI
            appendMessage(message, 'user');
            inputEl.value = '';
            inputEl.disabled = true; // Блокируем ввод во время запроса
            
            // Сохраняем в историю контекста
            aiChatHistory.push({ role: "user", parts: [{ text: message }] });
            
            showTypingIndicator();
            
            try {
                const responseText = await callGeminiAPI(aiChatHistory);
                hideTypingIndicator();

                console.error("Gemini API Error:", error);
                // Удаляем последнее сообщение пользователя из истории в случае сбоя
                aiChatHistory.pop();
            } finally {
                inputEl.disabled = false;
                inputEl.focus();
            }
        }

        // Вызов API с экспоненциальной задержкой (Exponential Backoff)
        async function callGeminiAPI(history) {
            const apiKey = ""; // API ключ пробрасывается средой автоматически
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;
            
            const payload = {
                contents: history,
                systemInstruction: {
                    parts: [{ text: systemInstruction }]
                }
            };
            
            let retries = 3;
            let delay = 1000;
            
            while (retries > 0) {
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                
                if (response.ok) {
                    const result = await response.json();
                    if (result.candidates && result.candidates[0].content && result.candidates[0].content.parts) {
                        return result.candidates[0].content.parts[0].text;
                    }
                    throw new Error("Неожиданная структура ответа API");
                } else if (response.status === 429) {
                    // Если достигнут лимит запросов — ждем и повторяем
                    await new Promise(r => setTimeout(r, delay));
                    delay *= 2;
                    retries--;
                } else {
                    throw new Error(`Ошибка API: ${response.status}`);
                }
            }
            throw new Error("Превышено максимальное количество попыток подключения");
        }
